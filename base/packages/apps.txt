# PurmaLinux - User Applications
# Aplicaciones de usuario que deben estar instaladas y preconfiguradas
# 
# Formato:
# nombre-paquete                    # Descripción breve
# nombre-paquete [CONFIG]           # Requiere configuración personalizada
# nombre-paquete [FLATPAK]          # Instalar vía Flatpak
# nombre-paquete [AUR/PPA]          # Requiere repositorio adicional
#
# Las configuraciones personalizadas van en: base/configs/apps/
